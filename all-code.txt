// Копия скрипта из Google Таблицы «красота» (Расширения → Apps Script).
// Исправлено: поиск строки по колонке key/url и normalizeKey без URL() (в Apps Script его нет).
// Добавлено: тематики (?sheets=1, ?sheet=, body.sheet), меню «Обзвон», formatAll() — оформление всех листов.

const SHEET_NAME = 'contacts'; // ⚠️ точное имя вкладки; регистр важен

const COL = {
  site:     'сайт',
  phone:    'телефон',
  result:   'результат',
  lpr:      'лпр',
  note:     'заметка',
  called:   'звонок',
  updated:  'обновлено'
};

/* ---------- GET: отдаём все строки как JSON(P) ---------- */
function doGet(e) {
  const p = (e && e.parameter) || {};
  // список вкладок для выбора тематики на сайте
  if (p.sheets) {
    const names = SpreadsheetApp.getActiveSpreadsheet().getSheets()
      .filter(s => !s.isSheetHidden()).map(s => s.getName());
    return out({ sheets: names }, e);
  }
  const sh = getSheet(p.sheet);
  if (!sh) return out({ error: 'Лист не найден: ' + (p.sheet || SHEET_NAME) }, e);

  const values = sh.getDataRange().getValues();
  if (values.length < 2) return out([], e);

  const headers = values.shift().map(h => String(h).trim());
  const rows = values.map(r => {
    const o = {};
    headers.forEach((h, i) => o[h] = r[i]);
    return o;
  });
  return out(rows, e);
}

/* ---------- POST: пишем результат/заметку/ЛПР/звонок ---------- */
function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const sh = getSheet(body.sheet);
    if (!sh) return out({ error: 'Лист не найден' }, e);

    let values = sh.getDataRange().getValues();
    let headers = values[0].map(h => String(h).trim());

    let iSite = headers.indexOf('key');
    if (iSite < 0) iSite = headers.indexOf(COL.site);
    if (iSite < 0) iSite = headers.indexOf('url');
    if (iSite < 0) return out({ error: 'Нет колонки «сайт»' }, e);

    // авто-создание всех недостающих колонок (включая лпр и звонок)
    const need = [COL.result, COL.note, COL.lpr, COL.called, COL.updated];
    let added = false;
    need.forEach(name => {
      if (headers.indexOf(name) < 0) {
        headers.push(name);
        sh.getRange(1, headers.length).setValue(name);
        added = true;
      }
    });
    if (added) values = sh.getDataRange().getValues(); // перечитываем после расширения

    const idx = name => headers.indexOf(name) + 1;

    // ищем строку по ключу (домен+путь, без параметров)
    const key = normalizeKey(body.key);
    for (let r = 1; r < values.length; r++) {
      if (normalizeKey(values[r][iSite]) === key) {
        const row = r + 1;
        if (body.result   !== undefined) sh.getRange(row, idx(COL.result)).setValue(body.result   || '');
        if (body.note     !== undefined) sh.getRange(row, idx(COL.note)).setValue(body.note       || '');
        if (body.lpr      !== undefined) sh.getRange(row, idx(COL.lpr)).setValue(body.lpr         || '');
        if (body.calledAt) {
          sh.getRange(row, idx(COL.called)).setValue(new Date(body.calledAt));
        }
        sh.getRange(row, idx(COL.updated)).setValue(new Date());
        return out({ ok: true, row }, e);
      }
    }
    return out({ error: 'Строка не найдена: ' + key }, e);
  } catch (err) {
    return out({ error: String(err) }, e);
  }
}

/* ---------- helpers ---------- */
function getSheet(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (name) return ss.getSheetByName(String(name)); // вкладка тематики («Еда», «Другое»…)
  return ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
}

function normalizeKey(url) {
  // в Apps Script нет URL(), разбираем вручную: домен + путь без параметров
  let s = String(url || '').trim().toLowerCase();
  s = s.replace(/^https?:\/\//, '').replace(/[?#].*$/, '').replace(/\/+$/, '');
  return s;
}

function out(payload, e) {
  const json = JSON.stringify(payload);
  const cb = e && e.parameter && e.parameter.callback;
  if (cb) {
    return ContentService.createTextOutput(cb + '(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

/* ================= ОФОРМЛЕНИЕ ТАБЛИЦЫ =================
   Меню «Обзвон → Оформить все листы» (или выбрать formatAll → ▶ Выполнить).
   Данные не меняются, только вид. Оформление ставится на весь лист целиком,
   поэтому новые строки от программы Search сразу выглядят так же. */
const STATUSES = ['Не взяли', 'Перезвонить', 'Интерес', 'Отказ'];
const STATUS_COLORS = {
  'Не взяли':    ['#e4ecfb', '#2c5bb8'],
  'Перезвонить': ['#fbf0dc', '#9a5f08'],
  'Интерес':     ['#e3f4ea', '#167a4a'],
  'Отказ':       ['#fae5e3', '#b0352f']
};
// цвет ярлычка вкладки по тематике
const TAB_COLORS = {
  'contacts': '#b4426b', 'Еда': '#e07a2f', 'Авто': '#3c6fd1', 'Спорт': '#2f9d62',
  'Медицина': '#2f8f83', 'Обучение и дети': '#7b5cc4', 'Бытовые услуги': '#8a6d3b', 'Другое': '#8c8794'
};
// подсказки к заголовкам (заголовки служебные — их читают сайт и программа Search)
const HEAD_NOTES = {
  'key': 'Ключ строки (ссылка без https). Нужен сайту, не менять.',
  'name': 'Название', 'category': 'Категория', 'address': 'Адрес', 'phones': 'Телефоны',
  'note': 'Сторона · округ Москвы', 'url': 'Ссылка на карточку / сайт',
  'результат': 'Статус звонка: Не взяли / Перезвонить / Интерес / Отказ',
  'заметка': 'Заметка с сайта-обзвонника (и отметки «WhatsApp: … 16.11»)',
  'лпр': 'Кто принимает решение', 'звонок': 'Когда звонили', 'обновлено': 'Когда меняли строку'
};

function onOpen() {
  SpreadsheetApp.getUi().createMenu('Обзвон')
    .addItem('Оформить все листы', 'formatAll')
    .addItem('Оформить текущий лист', 'formatCurrent')
    .addToUi();
}

function formatAll() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ss.getSheets().filter(sh => isCallSheet_(sh)).forEach(sh => formatSheet(sh));
  ss.toast('Все листы оформлены', 'Обзвон', 4);
}
function formatCurrent() { formatSheet(SpreadsheetApp.getActiveSheet()); }

// лист обзвона = есть колонка key/url и phones
function isCallSheet_(sh) {
  if (sh.getLastRow() < 1 || sh.getLastColumn() < 1) return false;
  const h = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(x => String(x).trim());
  return (h.indexOf('key') >= 0 || h.indexOf('url') >= 0) && h.indexOf('phones') >= 0;
}

function formatSheet(sh) {
  sh = sh || getSheet();
  const head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(h => String(h).trim());
  // во 2-й колонке стоял второй «key» с названием — это «name»
  if (head[0] === 'key' && head[1] === 'key') { sh.getRange(1, 2).setValue('name'); head[1] = 'name'; }
  [COL.result, COL.note, COL.lpr, COL.called, COL.updated].forEach(name => {
    if (head.indexOf(name) < 0) { head.push(name); sh.getRange(1, head.length).setValue(name); }
  });
  const cols = head.length;
  if (sh.getMaxColumns() > cols) sh.deleteColumns(cols + 1, sh.getMaxColumns() - cols);
  if (sh.getMaxRows() < sh.getLastRow() + 200) sh.insertRowsAfter(sh.getMaxRows(), sh.getLastRow() + 200 - sh.getMaxRows());
  const n = sh.getMaxRows() - 1;          // весь лист: новые строки тоже будут оформлены
  const idx = name => head.indexOf(name) + 1;

  sh.getRange(1, 1, sh.getMaxRows(), cols).setFontFamily('Manrope');
  sh.getRange(1, 1, 1, cols)
    .setFontColor('#ffffff').setFontWeight('bold').setFontSize(10)
    .setVerticalAlignment('middle').setHorizontalAlignment('left')
    .setNotes([head.map(h => HEAD_NOTES[h] || '')]);
  sh.setRowHeight(1, 40);
  sh.setFrozenRows(1);
  sh.setFrozenColumns(idx('name') || 0);

  const body = sh.getRange(2, 1, n, cols);
  body.setFontSize(11).setFontColor('#1b1820').setFontWeight('normal').setVerticalAlignment('middle')
      .setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP)
      .setBorder(null, null, true, null, null, true, '#ebe6e0', SpreadsheetApp.BorderStyle.SOLID);
  sh.setRowHeightsForced(2, n, 34);
  sh.getBandings().forEach(b => b.remove());
  sh.getRange(1, 1, n + 1, cols).applyRowBanding(SpreadsheetApp.BandingTheme.LIGHT_GREY, true, false)
    .setHeaderRowColor('#1b1820').setFirstRowColor('#ffffff').setSecondRowColor('#f7f5f2');

  const style = (name, width, fn) => { const c = idx(name); if (!c) return; sh.setColumnWidth(c, width); if (fn) fn(sh.getRange(2, c, n, 1)); };
  style('name', 260, r => r.setFontWeight('bold'));
  style('category', 150, r => r.setFontColor('#7a7380').setFontSize(10));
  style('address', 280, r => r.setFontColor('#7a7380').setFontSize(10));
  style('phones', 210, r => r.setFontWeight('bold').setFontColor('#1f9d62').setNumberFormat('@'));
  style('note', 150, r => r.setFontColor('#9a5f08').setFontSize(10));
  style('url', 220, r => r.setFontColor('#b4426b').setFontSize(10));
  style(COL.result, 130, r => r.setHorizontalAlignment('center').setFontWeight('bold'));
  style(COL.lpr, 160);
  style(COL.note, 340, r => r.setFontSize(10));
  style(COL.called, 110, r => r.setNumberFormat('dd.MM HH:mm').setFontColor('#7a7380').setHorizontalAlignment('center'));
  style(COL.updated, 110, r => r.setNumberFormat('dd.MM HH:mm').setFontColor('#7a7380').setHorizontalAlignment('center'));
  ['key', 'phone_1', 'phone_2', 'phone_3'].forEach(name => { const c = idx(name); if (c) sh.hideColumns(c); });

  const rules = [];
  const cRes = idx(COL.result);
  if (cRes) {
    const resRange = sh.getRange(2, cRes, n, 1);
    resRange.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(STATUSES, true).setAllowInvalid(true).build());
    STATUSES.forEach(s => rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(s)
      .setBackground(STATUS_COLORS[s][0]).setFontColor(STATUS_COLORS[s][1]).setRanges([resRange]).build()));
    const L = colLetter_(cRes), nc = idx('name') || 1;
    rules.push(SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=$' + L + '2="Отказ"')
      .setFontColor('#a49dab').setStrikethrough(true).setRanges([sh.getRange(2, nc, n, 1)]).build());
    rules.push(SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=$' + L + '2="Интерес"')
      .setBackground('#e3f4ea').setFontColor('#167a4a').setRanges([sh.getRange(2, nc, n, 1)]).build());
  }
  const cMsg = idx(COL.note);
  if (cMsg) rules.push(SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=REGEXMATCH($' + colLetter_(cMsg) + '2;"(WhatsApp|Telegram|Max): ")')
    .setFontColor('#2c5bb8').setRanges([sh.getRange(2, cMsg, n, 1)]).build());
  const cPh = idx('phones'), cNm = idx('name');
  if (cPh && cNm) rules.push(SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($' + colLetter_(cNm) + '2<>"";$' + colLetter_(cPh) + '2="")')
    .setBackground('#fff6e6').setRanges([sh.getRange(2, cPh, n, 1)]).build());
  sh.setConditionalFormatRules(rules);

  if (sh.getFilter()) sh.getFilter().remove();
  sh.getRange(1, 1, n + 1, cols).createFilter();
  if (TAB_COLORS[sh.getName()]) sh.setTabColor(TAB_COLORS[sh.getName()]);
}

function colLetter_(c) { let s = ''; while (c > 0) { const m = (c - 1) % 26; s = String.fromCharCode(65 + m) + s; c = (c - m - 1) / 26; } return s; }

===== ./apps-script/README.md =====
# Скрипт Google Таблицы

Чтобы статусы с сайта («Не взяли», «Перезвонить», «Интерес», «Отказ»), заметки и время звонка
записывались в таблицу и были видны всем, а сама таблица выглядела аккуратно.

## Установка (один раз, ~5 минут)

1. Откройте Google Таблицу → **Расширения → Apps Script**.
2. Замените содержимое `Code.gs` на [Code.gs](Code.gs) → 💾 Сохранить.
   (Если в старом скрипте было что-то своё, кроме чтения/записи для сайта, — сохраните его копию.)
3. Выберите функцию **formatSheet** → **▶ Выполнить** → разрешите доступ.
   Таблица получит тёмную шапку, зебру, ширину колонок, выпадающий список статусов с цветами,
   фильтр; служебные колонки key / phone_1–3 скроются. Данные не меняются.
4. **Начать развёртывание → Управление развёртываниями** → ✏️ → Версия: **Новая версия** → **Развернуть**.
   Ссылка `/exec` остаётся прежней. Доступ: «Выполнять от имени: я», «Кто имеет доступ: все».

## Что пишется в таблицу

| колонка     | что                                   |
|-------------|----------------------------------------|
| результат   | статус из выпадающего списка          |
| заметка     | ЛПР, время, что сказали               |
| звонок      | когда нажали «Позвонить»               |
| обновлено   | когда последний раз меняли строку      |

Статус можно менять и прямо в таблице — сайт подхватит при обновлении.

/* ---------- PICKER (нижняя шторка 1-2-3) ---------- */
function openPhoneSheet(id, action){
  const it=items.find(x=>x.id===id); if(!it) return;
  let options=it.phones.map((p,j)=>({phone:p,idx:j}));
  if(action!=='call') options=options.filter(o=>isMobile(digits(o.phone)));
  if(!options.length) return;

  // один вариант — сразу действие
  if(options.length===1){ performPhoneAction(id, action, options[0].idx); return; }

  const titles={call:'Кому позвонить?',wa:'WhatsApp — на какой номер?',
                tg:'Telegram — на какой номер?',max:'MAX — на какой номер?'};
  $('sheet').dataset.id=id;
  $('sheetTitle').textContent=titles[action]||'Выберите номер';
  const cur=(state[id]||{}).p||0;

  $('sheetList').innerHTML=options.map(o=>`
    <a class="sheet-item ${o.idx===cur?'sel':''}" data-sheet-idx="${o.idx}" href="${esc(phoneHref(action,digits(o.phone)))}"${action==='call'?'':' target="_blank" rel="noopener"'}>
      <span class="sheet-num">${o.idx+1}</span>
      <span class="sheet-val">${esc(fmt(o.phone))}</span>
    </a>`).join('');

  $('sheet').dataset.id=id;
  $('sheet').dataset.action=action;
  $('sheet').classList.add('show');
}
function closeSheet(){
  $('sheet').classList.remove('show');
  $('sheet').dataset.id='';
  $('sheet').dataset.action='';
}

/* прямая ссылка на номер: tel:, WhatsApp, Telegram, MAX */
function phoneHref(action,d){
  if(action==='call') return 'tel:+'+d;
  const id=$('sheet').dataset.id;
  return msgHref(action,d,items.find(x=>x.id===id));
}

function performPhoneAction(id, action, idx, viaLink){
  const it=items.find(x=>x.id===id); if(!it) return;
  const phone=it.phones[idx]; if(!phone) return;
  const d=digits(phone);
  const s=st(id); s.p=idx;

  if(action==='call'){
    s.calledAt=Date.now(); saveState();

/* ссылка на чат: WhatsApp — с текстом; Telegram — по номеру (текст в буфер);

function msgHref(action,d,it){
  if(!isBeautyTheme())it=null;                 // шаблоны — только для бьюти

  if(action==='max')return 'https://web.max.ru/';

function copyText(text){
  const legacy=()=>{
    const ta=document.createElement('textarea');
    ta.value=text;ta.setAttribute('readonly','');ta.style.cssText='position:fixed;top:-1000px;opacity:0';
    document.body.appendChild(ta);ta.select();

  if(navigator.clipboard&&window.isSecureContext){
    return navigator.clipboard.writeText(text).then(()=>true,()=>legacy());

}

/* Построение списка и HTML карточек */
function build(rows){
  const seen=new Map();
  rows.forEach((row,i)=>{
    const site=pick(row,/^(сайт|site|url)$/i)||pick(row,/сайт|site|url/i);
    if(!site)return;
    const key=keyOf(site);
    if(seen.has(key))return;

    const c=CONTACTS[key]||{};
    // в бьюти-вкладку случайно попало не бьюти (кафе, гостиница…) — не показываем
    if(isBeautyTheme()&&!c.name&&!isBeautyCategory(pick(row,/^(category|категория)$/i)))return;
    const sheetPhone=pick(row,/^телефон|phone/i);
    // разбиваем склеенные номера из таблицы и из contacts.js, дедуп
    const fromSheet=splitPhones(sheetPhone);
    const fromContacts=splitPhones(c.phones||[]);
    const phones=[...new Set([...fromSheet,...fromContacts])];

    const sheetRes = pick(row,/результат/i);
    const sheetNote= pick(row,/заметка/i);
    const sheetCall= pick(row,/^звонок|called/i);
    const sheetUpd = pick(row,/^обновлено|updated/i);
    const resKey=Object.keys(STATUS).find(k=>STATUS[k].t.toLowerCase()===sheetRes.toLowerCase())||'';
    const calledAt=sheetCall?new Date(sheetCall).getTime():0;

    const local=state[key]||{};
    state[key]={
      s:       local.s||resKey||'',
      n:       local.n||sheetNote||'',
      calledAt:local.calledAt||calledAt||0,
      ts:      Math.max(local.ts||0,sheetUpd?new Date(sheetUpd).getTime():0),
      p:       local.p||0
    };

    if(local.s&&!resKey)        queuePush(key,{result:STATUS[local.s].t,note:local.n||'',calledAt:local.calledAt||0});
    else if(local.n&&!sheetNote)queuePush(key,{result:local.s?STATUS[local.s].t:'',note:local.n,calledAt:local.calledAt||0});
    else if(local.calledAt&&!calledAt)queuePush(key,{result:local.s?STATUS[local.s].t:'',note:local.n||'',calledAt:local.calledAt});

    // строки из Яндекс Карт (davydovserg/Search): название в «key» (или «name»), район в «note»
    const rowKey =pick(row,/^key$/i);
    const rowName=pick(row,/^(name|название)$/i)||(/^(https?:\/\/|[\w-]+\.[\w.-]+\/)/i.test(rowKey)?'':rowKey);
    const name=cleanName(c.name||rowName,key);
    const area=c.name?'':pick(row,/^note$/i);
    const sub=[c.category||pick(row,/^(category|категория)$/i),c.address||pick(row,/^(address|адрес)$/i),area].filter(Boolean).join(' · ');
    seen.set(key,{
      id:key,i,site,phones,
      name,
      sub,
      note:c.note||'',
      search:(name+' '+sub+' '+key+' '+phones.map(p=>digits(p)+' '+fmt(p)).join(' ')).toLowerCase()
    });
  });
  items=[...seen.values()];
  saveState();
  render();
}

/* Очередь изменений → Google Sheets (POST no-cors, keepalive) */
const outbox=new Map();
let outTimer=null;

function queuePush(key,patch){
  const cur=outbox.get(key)||{};
  outbox.set(key,Object.assign(cur,patch));
  clearTimeout(outTimer);
  outTimer=setTimeout(flushOutbox,900);
}

async function flushOutbox(){
  if(!outbox.size)return;
  const entries=[...outbox.entries()];
  const sheet=isBeautyTheme()?undefined:theme;   // вкладка на момент отправки
  for(const [key,patch] of entries){
    const body=JSON.stringify({
      key,
      sheet,
      result:   patch.result,
      note:     patch.note,
      calledAt: patch.calledAt
    });
    try{
      const blob=new Blob([body],{type:'text/plain;charset=UTF-8'});
      const resp=await fetch(API_URL,{method:'POST',mode:'no-cors',keepalive:true,body:blob});
      if(resp.type==='opaque'||resp.type==='basic')outbox.delete(key);
    }catch(e){
      console.warn('sync failed, will retry:',key,e);
    }
  }
}

function jsonp(params,cb,timeout){
  const name='jsonp_'+Date.now()+'_'+Math.floor(Math.random()*1e6);
  let done=false;
  const t=setTimeout(()=>{if(!done){done=true;delete window[name];cb(null)}},timeout||15000);
  window[name]=d=>{if(done)return;done=true;clearTimeout(t);delete window[name];cb(d)};
  const s=document.createElement('script');
  s.src=API_URL+'?callback='+name+'&'+params+'&t='+Date.now();
  s.onerror=()=>{if(!done){done=true;clearTimeout(t);delete window[name];cb(null)}};

}

/* Разбивает строку/массив на отдельные телефоны.
   Понимает склеенные: "+791001900517985121201179031558860" → 3 номера. */
function splitPhones(input){
  if(input==null) return [];
  if(Array.isArray(input)){
    const out=[];
    input.forEach(v=>out.push(...splitPhones(v)));
    return [...new Set(out.filter(Boolean))];
  }
  const s=String(input).trim();
  if(!s) return [];
  const re=/(?:\+?7|8)[\s\-()]?\d{3}[\s\-()]?\d{3}[\s\-]?\d{2}[\s\-]?\d{2}/g;
  const found=s.match(re);
  if(found&&found.length) return [...new Set(found.map(p=>'+'+digits(p)))];
  const d=digits(s);
  if(d.length===10||d.length===11) return ['+'+d];
  return [];
}

const VERSION = 5;                       // ← должно совпадать с manifest.json.version
const CACHE_NAME = 'krasota-v' + VERSION;

const ASSETS = [
  './',
  './index.html',
  './icons/icon.svg',
  // './css/layout.css', './css/cards.css', './css/controls.css', './js/app.js'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then((cache) => Promise.allSettled(ASSETS.map((u) => cache.add(new Request(u, { cache: 'reload' })))))
      .then(() => self.skipWaiting())
  );
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys()
      .then((keys) => Promise.all(
        keys.filter((k) => k !== CACHE_NAME).map((k) => caches.delete(k))
      ))
      .then(() => self.clients.claim())
  );
});

// ── Обмен сообщениями с клиентом ──────────────────
self.addEventListener('message', (event) => {
  if (!event.data) return;
  if (event.data.type === 'SKIP_WAITING') return self.skipWaiting();
  if (event.data.type === 'GET_VERSION' && event.ports[0]) {
    event.ports[0].postMessage({ version: VERSION });
  }
});

// ── fetch ─────────────────────────────────────────
self.addEventListener('fetch', (event) => {
  if (event.request.method !== 'GET') return;

  const url = new URL(event.request.url);

  // manifest.json — всегда из сети, иначе не увидим новую версию
  if (url.pathname.endsWith('/manifest.json') || url.pathname.endsWith('manifest.json')) {
    event.respondWith(fetch(event.request, { cache: 'no-store' }));
    return;
  }

  // страница, js и css — сначала сеть (иначе после деплоя застревает старый код), кэш — только офлайн
  if (url.origin === self.location.origin &&
      (event.request.mode === 'navigate' || /\.(html|js|css)$/.test(url.pathname))) {
    event.respondWith(
      fetch(event.request, { cache: 'no-cache' }).then((resp) => {
        if (resp.ok) {
          const copy = resp.clone();
          caches.open(CACHE_NAME).then((c) => c.put(event.request, copy));
        }
        return resp;
      }).catch(() => caches.match(event.request).then((c) => c || caches.match('./index.html')))
    );
    return;
  }

  event.respondWith(
    caches.match(event.request).then((cached) => {
      if (cached) return cached;
      return fetch(event.request).then((resp) => {
        if (resp.ok && url.origin === self.location.origin) {
          const copy = resp.clone();
          caches.open(CACHE_NAME).then((c) => c.put(event.request, copy));
        }
        return resp;
      }).catch(() => {
        if (event.request.mode === 'navigate') return caches.match('./index.html');
      });
    })
  );
});

MANIFEST_URL="${MANIFEST_URL:-https://goshva.github.io/krasota/manifest.json}"

# Тянем version из манифеста (в обход кэша GH Pages)
current_manifest_version() {
  local body v
  body="$(fetch "$MANIFEST_URL")" || return 1
  [[ -z "$body" ]] && return 1
  v="$(printf '%s' "$body" \
       | grep -oE '"version"[[:space:]]*:[[:space:]]*[0-9]+' \
       | grep -oE '[0-9]+$' || true)"
  [[ -z "$v" ]] && return 1
  printf '%s' "$v"
}

check_once() {
  local remote prev
  if ! remote="$(current_manifest_version)"; then
    log "⚠ манифест недоступен — пропуск"
    return 0
  fi

  if [[ ! -s "$STATE_FILE" ]]; then
    printf '%s' "$remote" > "$STATE_FILE"
    log "baseline: v${remote}"
    return 0
  fi

  prev="$(tr -dc '0-9' < "$STATE_FILE" || echo 0)"
  [[ -z "$prev" ]] && prev=0

  if (( remote > prev )); then
    printf '%s' "$remote" > "$STATE_FILE"
    local gh; gh="$(gh_actions_status || true)"
    notify "🚀 <b>Новая версия v${remote}</b> (было v${prev})
Проверить: ${URL}
Время: $(date '+%F %T')${gh:+
CI: <code>${gh}</code>}"
  else
    log "без изменений (v${remote})"
  fi
}
